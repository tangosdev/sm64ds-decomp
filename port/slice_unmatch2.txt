# ---- 1. DA_PG_DFDR (ov027), +2 --------------------------------------------
# src/func_ov027_02111d38.cpp and src/func_ov027_02111cfc.cpp both TAIL JUMP
# through the ROM's own record:
#
#     mov  eax, DWORD PTR _c$[ebp]
#     mov  edx, DWORD PTR [eax+972]     ; c->pp, the ROM's +0x3cc
#     mov  ecx, DWORD PTR [edx+4]       ; the adjustment word  (cfc: [edx+12])
#     add  ecx, eax
#     mov  eax, DWORD PTR [edx]         ; the code word        (cfc: [edx+8])
#     pop  ebp
#     jmp  eax
#
# so with /vmg /vmm MSVC's member pointer IS the ROM's 8-byte pair and `c->pp+1`
# IS the element's second pair. What was missing is the SEAT: nothing rewrote
# data_ov027_02113ce4, so the table held DS code addresses and a matched
# dispatcher would have jumped to 0x02111ca8 -- reserved zeroed main RAM on this
# host. port/unmatched/DaPgDfdr_StateDispatch.cpp is now that seat: it rewrites
# the four .data SOURCE pairs with host bodies before
# __sinit_ov027_02112df8 copies them, the gate-178 ordering, called from
# hal/actor_overlays.cpp. All four pairs re-read out of
# extracted/overlays/overlay_0027.bin at (addr - 0x021111a0), every adjustment
# word ROM ZERO:
#   data_ov027_021139d4 {02111ca8,0}  data_ov027_021139cc {02111c48,0}
#   data_ov027_021139dc {02111b2c,0}  data_ov027_021139c4 {02111a28,0}
# REFERRER: the ROM's own setter func_ov027_02111d70 tail-jumps into
# func_ov027_02111d38 after storing the element pointer at +0x3cc, and
# daPgDfdr_c's Behavior reaches func_ov027_02111cfc; both are called by C name
# from the rest of the ov027 cast, which is why the two symbols never left the
# link -- only their bodies did.
src/func_ov027_02111d38.cpp

# and the ROM's two pairs at [edx]/[edx+4] and [edx+8]/[edx+12]. Their name

# __sinit_ov002_02107118 runs from the harness at startup, long before any
# actor-overlay pass, and data_ov002_0210ad38 is mounted one pair wide. Called

# construction: a YoshiEgg only exists inside a level. All eight source pairs
# re-read out of extracted/overlays/overlay_0002.bin at (addr - 0x020ad660),

# ---- 3. THE PAINTING (ov080), +3 ------------------------------------------
# daPicGate_c's three per-mode dispatchers. The old host copy's header said
# MSVC "widens the PMF to the 16-byte general form" and therefore strides 0x20
# where the ROM strides 0x18. That is no longer true under /vmg /vmm and the
# listing says so: src/func_ov080_02126ca0.cpp emits
#
#     lea eax, DWORD PTR [eax+eax*2]                      ; * 3
#     lea eax, ?data_ov080_02128628@@3PAUDisp@@A[eax*8]   ; * 8  == 0x18
#
# -- the ROM's own stride -- and _02126c60 / _02126c20 read the element's
# second and third pairs at [edx+8]/[edx+12] and [edx+16]/[edx+20], which is
# the ROM's {setup, update, render} record exactly.
#
# WHAT WAS ACTUALLY WRONG IS THE CALL, and it is the one place in the port
# where the member-pointer family does not tail-jump. All three TUs end
# `return 1;` after the dispatch, so MSVC cannot reuse the frame and emits
# `call eax` with the receiver in ecx and NOTHING pushed. Every other seated
# table in this tree relies on the tail jump leaving the dispatcher's own
# receiver argument at [esp+4] for a cdecl body to read. So the twelve seated
# Painting records now hold a one-line __fastcall FACE each
# (hal/actor_overlays.cpp, beside g_painting_states); __fastcall reads the
# receiver from ecx and returns `ret 0`, which balances a `call` and equally
# survives a tail jump. No body changes and no ROM word moves.
# The two mangled names InitResources asks for are bridged in the new
# port/hal/unmatch_aliases.cpp.
# REFERRER: _ZTV11Painting slots 0/6/9 (InitResources, Behavior, Render) --
# the three rows SEAT3's census records as PMF EXCEPTION for ov080
# (data_ov080_021282b4 slots 0, 6 and 9).
src/func_ov080_02126ca0.cpp
src/func_ov080_02126c60.cpp
src/func_ov080_02126c20.cpp
