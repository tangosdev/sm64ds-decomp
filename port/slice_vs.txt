# The lobby's start conductor is func_ov075_02118a84 (matched src): it waits

#                                            SaveData::SetDefaultValues(active)

#         data_ov075_0211c6ec[data_0209b2e4[0]], 0, 2, 0);

#     data_02092778 = 1;
#
# So the ROM stages the map IN THE SAME FRAME as the scene-3 request:
# "staged" concretely is data_02092110 = the map's level id (entrance 0,

# THE VS MAP LIST lives in ov075's .data: data_ov075_0211c6ec, four signed
# bytes {0x33, 0x2b, 0x1d, 0x2a} = levels 51, 43, 29, 42, indexed by
