cmake_minimum_required(VERSION 3.15)
project(naming_lab C)
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The game is 32-bit: configure with an x86 toolchain (vcvars32).")
endif()
add_library(naming_lab SHARED naming_lab.c)
set_property(TARGET naming_lab PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
target_compile_options(naming_lab PRIVATE /W3)
target_link_libraries(naming_lab PRIVATE user32 ws2_32)
