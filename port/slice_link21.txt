#     __ZN12dScMgSlot1_c8BehaviorEv      ?Behavior@dScMgSlot1_c@@UAEHXZ
#     __ZN11dScMgJump_c13OnYoshiTryEatEi ?OnYoshiTryEat@dScMgJump_c@@UAEXH@Z

# needs: tools/enroll.py leaves a bannered draft out of the ROM build, so the
# byte gate keeps the cartridge's own bytes for both ranges.

#     src/_ZN12dScMgSlot1_c8BehaviorEv.cpp       ov006 0x0210c9e0, size 0x81c
#                                                vtable slot 6
